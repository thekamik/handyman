Podpowiedź 1 Funkcja request(flow) w addonie działa, zanim żądanie opuści proxy. Pola formularza zmieniasz przez flow.request.urlencoded_form, a podpis policzysz modułami hmac i hashlib.
Podpowiedź 2 
import hashlib, hmac

KEY = b"{{ key }}"

def request(flow):
    if flow.request.method == "POST" and flow.request.path.startswith("/exercise15/"):
        form = flow.request.urlencoded_form
        form["cmd"] = "unlock"
        message = f"{form['nonce']}:unlock".encode()
        form["sig"] = hmac.new(KEY, message, hashlib.sha256).hexdigest()
