W mitmproxy klawisz x eksportuje żądanie np. jako komendę curl, co pomaga przepisać je na Pythona.
    Podpowiedź 2 
    import re
import requests

BASE = "http://192.168.1.10:8000/exercise16"   # address of the server
PROXIES = {"http": "http://127.0.0.1:8080"}     # mitmproxy in regular mode
# Reverse mode: BASE = "http://127.0.0.1:8080/exercise16" and PROXIES = {}

s = requests.Session()
s.proxies = PROXIES
page = s.get(f"{BASE}/").text
csrf = re.search(r'name="csrf" value="(\w+)"', page).group(1)
s.post(f"{BASE}/login", data={"csrf": csrf, "username": "{{ username }}", "password": "{{ password }}"})
for n in range(1, {{ pages }} + 1):
    for entry in s.get(f"{BASE}/api/logs", params={"page": n}).json()["entries"]:
        if "handyman{" in entry:
            print(n, entry)
