import requests

BASE = "http://192.168.1.10:8000/exercise17"
PROXIES = {"http": "http://127.0.0.1:8080"}

challenge = requests.get(f"{BASE}/api/challenge", proxies=PROXIES).json()
answer = {"token": challenge["token"], "sum": sum(challenge["numbers"])}
print(requests.post(f"{BASE}/api/answer", json=answer, proxies=PROXIES).json())
